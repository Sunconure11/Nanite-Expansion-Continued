
namespace = nano_colossus
# Transmuted
# 4.5: planet killer destruction events use carrier scope
# (mirrors vanilla planet_destruction.100 in events\planet_destruction_events.txt)
carrier_event = {
	id = nano_colossus.1300
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		# Don't fire the narrative event if the planet is destroyed by an arkship (4.5)
		if = {
			limit = {
				from = {
					is_arkship_ship = no
				}
			}
			if = {
				limit = { merg_is_frozen = yes }
				from.owner = {
					country_event = { id = nano_colossus.1301 }
				}
			}
			else_if = {
				limit = {
					NOT = { is_planet_class = pc_ai }
				}
				from.owner = {
					country_event = { id = nano_colossus.1301 }
				}
			}
		}
		# Dismantle observation post
		if = {
			limit = { has_observation_outpost = yes }
			observation_outpost_owner = {
				country_event = { id = planet_destruction.1000 days = 1 }
			}
		}
		# Contingency Machine World
		if = {
			limit = {
				is_planet_class = pc_ai
				NOT = { has_planet_flag = machine_lair }
			}
			set_carrier_flag = destroyed_by_colossus
			set_carrier_flag = planet_cracked
			carrier_event = { id = crisis.2040 }
		}
		# Contingency Final Machine World
		if = {
			limit = {
				is_planet_class = pc_ai
				has_planet_flag = machine_lair
			}
			set_carrier_flag = destroyed_by_colossus
			set_carrier_flag = planet_cracked
			from.owner = { save_event_target_as = final_machine_world_destroyer }
			stop_crisis_sound = yes
			carrier_event = { id = crisis.2046 }
		}
		# Swarm Situation Log counter
		if = {
			limit = {
				exists = owner
				owner = { is_country_type = swarm }
			}
			every_country = {
				limit = { has_event_chain = "prethoryn_scourge_chain" }
				add_event_chain_counter = {
					event_chain = "prethoryn_scourge_chain"
					counter = "infested_worlds"
					amount = -1
				}
				add_event_chain_counter = {
					event_chain = "prethoryn_scourge_chain"
					counter = "infested_worlds_cleansed"
					amount = 1
				}
			}
		}
		if = {
			limit = {
				OR = {
					is_active_resolution = "resolution_rulesofwar_reverence_for_life"
					is_active_resolution = "resolution_rulesofwar_independent_tribunals"
					is_active_resolution = "resolution_rulesofwar_last_resort_doctrine"
					is_active_resolution = "resolution_rulesofwar_demobilization_initiative"
				}
			}
			from.owner = {
				set_timed_country_flag = { flag = resolution_breached_fired_cracker years = 10 }
			}
		}
		if = {
			limit = {
				exists = owner
				owner = {
					NOT = { is_same_empire = from.owner }
					is_default_or_fallen = yes
				}
			}
			# Generate threat
			if = {
				limit = { is_colony = yes }
				add_threat = { who = from.owner amount = 3 }
				# Modifier for allies + those upset by genocide
				every_country = {
					limit = {
						NOR = {
							is_same_empire = from.owner
							is_same_empire = root.owner
							AND = {
								has_federation = yes
								is_in_federation_with = from.owner
							}
						}
						OR = {
							has_communications = from.owner
							has_communications = root.owner
						}
						OR = {
							AND = {
								has_federation = yes
								is_in_federation_with = root.owner
							}
							has_ai_personality = awakened_fallen_empire_xenophile
							AND = {
								merg_is_default_empire = yes
								OR = {
									is_egalitarian = yes
									is_xenophile = yes
								}
							}
						}
					}
					if = {
						limit = { root = { is_colony = no } }
						add_opinion_modifier = { modifier = opinion_cracked_an_uninhabited_world who = from.owner }
					}
					else = {
						add_opinion_modifier = { modifier = opinion_cracked_a_world who = from.owner }
					}
				}
			}
			else = {
				add_threat = { who = from.owner amount = 1 }
			}
			# Modifiers for victim
			owner = {
				if = {
					limit = { root = { is_colony = no } }
					add_opinion_modifier = { modifier = opinion_cracked_my_uninhabited_world who = from.owner }
				}
				else = {
					add_opinion_modifier = { modifier = opinion_cracked_my_world who = from.owner }
				}
				if = {
					limit = { is_gestalt = no }
					add_modifier = { modifier = colossus_victim years = 30 }
				}
			}
			if = {
				limit = {
					exists = owner
					owner = { is_primitive = yes }
				}
				# Modifier for allies + those upset by genocide
				every_country = {
					limit = {
						NOT = { is_same_empire = from.owner }
						OR = {
							has_communications = from.owner
							has_communications = root.owner
						}
						OR = {
							has_ai_personality = awakened_fallen_empire_xenophile
							AND = {
								merg_is_default_empire = yes
								NOR = {
									is_xenophobe = yes
									is_homicidal = yes
									is_xenophile = yes
								}
							}
						}
					}
					add_opinion_modifier = { modifier = opinion_cracked_a_primitive_world who = from.owner }
				}
				every_country = {
					limit = {
						NOT = { is_same_empire = from.owner }
						OR = {
							has_communications = from.owner
							has_communications = root.owner
						}
						OR = {
							has_ai_personality = awakened_fallen_empire_xenophile
							AND = {
								merg_is_default_empire = yes
								is_xenophile = yes
							}
						}
					}
					add_opinion_modifier = {
						modifier = opinion_cracked_a_primitive_world_phile
						who = from.owner
					}
				}
			}
		}
		# Add war exhaustion to planet owner
		if = {
			limit = { exists = owner }
			owner = {
				if = {
					limit = { #isn't the case when the crisis bombards planets, for instance (4.5 vanilla guard)
						is_at_war_with = from.fleet.owner
					}
					add_static_war_exhaustion = {
						attacker = from.fleet.owner
						location = root
						value_for_planet_destruction = 1.0
					}
				}
			}
		}
		# 4.5: destroy the orbital defence fleet so it doesn't orbit a dead world
		if = {
			limit = {
				exists = orbital_defence
			}
			destroy_fleet = orbital_defence
		}
		remove_all_buildings = yes
		every_owned_pop_group = {
			kill_all_pop = yes	# 4.5: vanilla scripted effect; "amount = 100%" was never valid
		}
		destroy_colony = yes
		# Destroy habitats and ringworlds
		if = {
			limit = { merg_is_habitat = yes }
			clear_consecration_effect = yes	# 4.5: vanilla clears consecrations before cracking habitats
			spawn_habitat_cracker_effect = yes
			remove_planet = yes
		}
		else_if = {
			limit = { is_planet_class = pc_cosmogenesis_world }	# 4.5: Cosmogenesis lathe worlds
			spawn_lathe_cracker_effect = yes
		}
		else_if = {
			limit = { has_ringworld_output_boost = yes }
			spawn_ringworld_cracker_effect = yes
			change_pc = pc_ringworld_habitable_damaged
		}
		else_if = {
			limit = { habitable_planet = yes }
			change_pc = pc_gray_goo
			clear_deposits = yes
			add_modifier = { modifier = "terraforming_candidate" days = -1 }
			random_list = {
				1 = {
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
				}
				1 = {
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
				}
				1 = {
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
					add_deposit = d_nanites_deposit
				}
			}
		}
		else = {
			change_pc = pc_gray_goo
			clear_deposits = yes
			add_deposit = d_nanites_deposit
			add_deposit = d_nanites_deposit
			add_deposit = d_nanites_deposit
			add_deposit = d_nanites_deposit
		}
	}
}

country_event = {
	id = nano_colossus.1301
	title = nano_colossus.1301.name
	desc = nano_colossus.1301.desc
	picture = GFX_evt_gray_gooed_planet
	show_sound = event_mystic_reveal
	location = from
	is_triggered_only = yes

	trigger = {
		NOR = {
			has_country_flag = fired_deconstructor
			from = { is_artificial = yes }
		}
	}

	immediate = {
		set_country_flag = fired_deconstructor
	}

	option = {
		name = nano_colossus.1301.a
		trigger = {
			NOR = {
				has_valid_civic = civic_fanatic_purifiers
				has_valid_civic = civic_machine_terminator
			}
		}
	}
	option = {
		name = planet_destruction.101.a.purifexter
		trigger = {
			OR = {
				has_valid_civic = civic_fanatic_purifiers
				has_valid_civic = civic_machine_terminator
			}
		}
		add_monthly_resource_mult = {
			resource = unity
			value = 60
			min = 600
			max = 3000
		}
	}
}
